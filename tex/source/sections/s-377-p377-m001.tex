\PoemStart{s-377-p377-m001}{Der Übermensch.}
\PoemSubtitle{(Novalis an Nietzsche.)}
\begin{BookVerse}
\VerseLine{0}{Will sich Natur zur Ruhe sacht gewöhnen,}
\VerseLine{1}{Musst du die wilde Zeit zurückbeschwören?}
\VerseLine{1}{Wer mag wohl noch des Mammuths Wuthschrei hören,}
\VerseLine{0}{Die Saat zerstampft sehn von den Enakssöhnen!}
\Stanza
\VerseLine{0}{Nicht Schaffen ist das Letzte, noch Zerstören:}
\VerseLine{1}{Soll Ordnung nicht den Bau am Ende krönen?}
\VerseLine{0}{Und regt sich’s allerwärts vom sinnvoll Schönen,}
\VerseLine{1}{Meinst du, dass wir am Grossen viel verlören?}
\Stanza
\VerseLine{0}{Lass auch im Menschen, nach der Vorwelt Kämpfen,}
\VerseLine{0}{Des Wollens und Befehlens Gluth sich dämpfen:}
\VerseLine{0}{Was Übermensch! Ein Kind träumt Ungeheuer.}
\Stanza
\VerseLine{1}{Dein Held Prometheus, stahl er nicht das Feuer,}
\VerseLine{0}{Dass es, vertheilt auf hundert fleiss’ge Herde,}
\VerseLine{0}{Des Allzerschmett’rers Hand entwunden werde?}
\end{BookVerse}
\PoemEnd
