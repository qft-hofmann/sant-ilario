\PoemStart{s-379-p379-m002}{Der Philosoph heirathet.}
\begin{BookVerse}
\VerseLine{0}{Geist! auf Nimmerwiedersehn!}
\VerseLine{1}{Lebe wohl, den einst ich hatte!}
\VerseLine{1}{Habe Dank: als Ehegatte}
\VerseLine{0}{Muss ich ohne dich bestehn.}
\Stanza
\VerseLine{0}{Lass mich, wie das Schiff die Ratte,}
\VerseLine{1}{Ganz im Glücke untergehn.}
\VerseLine{0}{Geist, auf Nimmerwiedersehn!}
\VerseLine{1}{Lebe wohl, den einst ich hatte!}
\Stanza
\VerseLine{0}{Mag es zärtlich nun geschehn,}
\VerseLine{1}{Dass ich dich zur Ruh’ bestatte.}
\VerseLine{1}{In die Zukunft soll, die glatte,}
\VerseLine{0}{Nicht dein rauher Athem wehn.}
\VerseLine{0}{Geist! auf Nimmerwiedersehn!}
\end{BookVerse}
\PoemEnd
