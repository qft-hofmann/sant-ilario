\PoemStart{s-387-p387-m001}{Gedanken des Südens.}
\begin{BookVerse}
\VerseLine{0}{Wie nenn’ ich euch, ihr nicht mehr sehnsuchtkranken,}
\VerseLine{1}{Ihr in Erfüllung ruhevoll gesunden!}
\VerseLine{1}{Ihr, die im Sei’nden Sinn und Glück gefunden,}
\VerseLine{0}{Nicht fiebernd mehr Unmögliches umranken!}
\Stanza
\VerseLine{0}{Nicht mehr geschreckt von Bildern, die versanken,}
\VerseLine{0}{Nicht nachgepeinigt von vernarbten Wunden,}
\VerseLine{1}{Wie lasst ihr kindlich euch das Leben munden,}
\VerseLine{0}{Oh ihr, des Südens Sonnentraumgedanken!}
\Stanza
\VerseLine{0}{Oh Glück, das mir vom Himmel niederfiel!}
\VerseLine{1}{Oh reines Jetzt, mit keinem Einst gemischt!}
\VerseLine{0}{Wie still und glatt das Meer! hat wohl mein Kiel}
\Stanza
\VerseLine{1}{Die Furche, die er pflügte, selbst verwischt?}
\VerseLine{0}{Rings um mein Boot ein leises Wellenspiel,}
\VerseLine{1}{Das kaum sich regt und flüsternd schon erlischt \Dots{2}}
\end{BookVerse}
\PoemEnd
