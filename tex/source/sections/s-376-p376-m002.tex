\PoemStart{s-376-p376-m002}{Englischer Sonntag.}
\begin{BookVerse}
\VerseLine{0}{Dieses Glöckleins rastlos Lärmen,}
\VerseLine{1}{Ohrenmarter ohne Gleichen!}
\VerseLine{0}{Denen’s gilt, in dichten Schwärmen}
\VerseLine{1}{Sieht man sie zur Kirche schleichen.}
\Stanza
\VerseLine{0}{Wo an Gott wir frei uns wärmen,}
\VerseLine{1}{Denen muss der Pfarr ihn reichen.}
\VerseLine{0}{Dieses Glöckleins rastlos Lärmen,}
\VerseLine{1}{Ohrenmarter ohne Gleichen!}
\Stanza
\VerseLine{0}{Gähnen, Spleen, verdross’nes Härmen:}
\VerseLine{1}{Will der Tag nicht endlich weichen?}
\VerseLine{0}{Ach! auf dieser Schafe Därmen}
\VerseLine{1}{Könnt’ man Davids Psalmen streichen! —}
\VerseLine{0}{Dieses Glöckleins rastlos Lärmen!}
\end{BookVerse}
\PoemEnd
