\PoemStart{s-386-p386-m002}{Hellenismus.}
\begin{BookVerse}
\VerseLine{1}{Einerlei ist nicht genung.}
\VerseLine{0}{Selbst den Griechen schien’s von Nöthen,}
\VerseLine{0}{Durch ein Satyrspiel zu tödten}
\VerseLine{1}{Der Tragödie heil’gen Schwung.}
\Stanza
\VerseLine{1}{Posse und Begeisterung}
\VerseLine{0}{Muss der Mensch zusammenlöthen.}
\VerseLine{1}{Einerlei ist nicht genung,}
\VerseLine{0}{Frag’ die Griechen, frage Goethen.}
\Stanza
\VerseLine{0}{Lern’ von Vögeln Flug, von Kröten}
\VerseLine{1}{Kriechen und von Fröschen Sprung,}
\VerseLine{0}{Musisch harfen, faunisch flöten:}
\VerseLine{1}{Mensch! so bleibst du griechisch jung!}
\VerseLine{1}{Einerlei ist nicht genung.}
\end{BookVerse}
\PoemEnd
