\PoemStart{s-375-p375-m001}{Vorsommer.}
\begin{BookVerse}
\VerseLine{1}{Bist du es, Sommer, der sich so verfrüht?}
\VerseLine{0}{Flog ich so früh, so stürmisch dir entgegen?}
\VerseLine{1}{Hier treff’ ich und umarm’ dich, dass es sprüht,}
\VerseLine{0}{Du überschüttest mich mit deinem Segen.}
\Stanza
\VerseLine{0}{Hier will ich mich in deinen Athem legen,}
\VerseLine{1}{Der summend, sengend um mich webt und glüht:}
\VerseLine{1}{März ist es! und die ros’ge Mandel blüht,}
\VerseLine{0}{Aufspriesst es bunt und toll an allen Wegen.}
\Stanza
\VerseLine{0}{Der Alpsee giert, ein glüh’nder Molochsbauch,}
\VerseLine{1}{Nach weisser Mädchen marmorkühlen Gliedern.}
\VerseLine{0}{Oh Lenz! willst du den Sommer überspringen?}
\Stanza
\VerseLine{0}{Die Traube reift dein ungestümer Hauch:}
\VerseLine{1}{Schon hör’ ich Winzer rufen und erwidern}
\VerseLine{0}{Und mit den Sicheln Winzermesser klingen.}
\end{BookVerse}
\PoemEnd
