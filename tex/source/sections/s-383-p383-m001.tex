\PoemStart{s-383-p383-m001}{Wortes Ohnmacht.}
\begin{BookVerse}
\VerseLine{0}{Scharf musternd späht mein Auge in die Menge:}
\VerseLine{1}{Nie bist es du, nach der das Herz begehrt.}
\VerseLine{1}{Nun ist es schmachtend innenwärts gekehrt,}
\VerseLine{0}{Ob selbst dein Bild zu malen ihm gelänge.}
\Stanza
\VerseLine{0}{Hier hilft das Ohr: im Fall und Fluss der Klänge}
\VerseLine{1}{Form’ ich dein tönend Gleichniss. Doch wer lehrt}
\VerseLine{1}{Mich Zartheit, die dich nicht verscheucht? wer wehrt}
\VerseLine{0}{Dem Wort, dass sich’s verletzend nah dir dränge?}
\Stanza
\VerseLine{0}{Oh Wort! in deiner Faust, der plump geballten,}
\VerseLine{1}{Wie liesse sich der leichte Falter greifen?}
\VerseLine{1}{Du kannst ihm nur den Staub vom Flügel streifen.}
\Stanza
\VerseLine{0}{Wie taugst du wenig, Sammt mit schweren Falten,}
\VerseLine{1}{Dich schleierweich um Lichtgestalt zu schmiegen!}
\VerseLine{1}{Du kannst als Bahrtuch nur auf Särgen liegen.}
\end{BookVerse}
\PoemEnd
