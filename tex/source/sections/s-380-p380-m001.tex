\PoemStart{s-380-p380-m001}{Brautnacht.}
\begin{BookVerse}
\VerseLine{0}{Was ist, Geliebte? Sammelst du die Welt}
\VerseLine{1}{Aus ihres Kreisens wirrverschlung’nem Reih’n?}
\VerseLine{1}{Dein Auge schwimmt: verfolgt es wohl den Stein,}
\VerseLine{0}{Der in den Born der Ewigkeiten fällt?}
\Stanza
\VerseLine{1}{Was ist? kannst du der Liebe nicht verzeihn,}
\VerseLine{0}{Auf deren letzten Grund wir uns gestellt?}
\VerseLine{1}{Du blickst so fremd: wie wenig muss es sein,}
\VerseLine{0}{Das dich Entrückte noch im Leben hält!}
\Stanza
\VerseLine{0}{Du blickst so fremd: wo weilen deine Sinne?}
\VerseLine{0}{Wen schaust du an? wem gabst du deine Minne?}
\VerseLine{1}{Zur Hölle, Weib, das mir die Ehe brach! —}
\Stanza
\VerseLine{0}{Sie wacht, sie lächelt mir, dem Reugequälten:}
\VerseLine{0}{„Dir, wie du warest, eh’ wir uns vermählten,}
\VerseLine{1}{Dem Fernen, der mich liebte, schau’ ich nach.“}
\end{BookVerse}
\PoemEnd
