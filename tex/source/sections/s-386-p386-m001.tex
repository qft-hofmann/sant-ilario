\PoemStart{s-386-p386-m001}{Ideale.}
\begin{BookVerse}
\VerseLine{0}{Die Bilder all, die uns zu Häupten hingen —}
\VerseLine{1}{Manch eines hängt noch, manches liegt in Scherben}
\VerseLine{1}{Doch Fleisch zu werden, blühend roth zu färben}
\VerseLine{0}{Den fahlen Leib, wollt’ keinem noch gelingen.}
\Stanza
\VerseLine{0}{Zu uns’rem Heil! Die Bilder sind’s von Dingen,}
\VerseLine{1}{An die der Ruf erging uns zu verderben!}
\VerseLine{1}{Süss wissen sie um uns’re Kraft zu werben,}
\VerseLine{0}{Bis wir, bethört, den Traum zu leben zwingen.}
\Stanza
\VerseLine{0}{Sehnt einem Ziel sich Segel zu und Steuer —}
\VerseLine{1}{Sturmwarnung ist’s, g’rad’ dorthin nicht zu streben!}
\VerseLine{0}{Lockt dich ein Schatz, den reichen zu erbeuten —}
\Stanza
\VerseLine{0}{Die Hand davon! Und flammt der Liebe Feuer:}
\VerseLine{1}{Nicht ohne dieses Weib mehr kann ich leben —}
\VerseLine{0}{Nicht mit ihm kannst du es! das will’s bedeuten.}
\end{BookVerse}
\PoemEnd
