\PoemStart{s-379-p379-m001}{Unio mystica.}
\begin{BookVerse}
\VerseLine{0}{Ich träum’ von Ehe, Einssein ohn’ Beschränken,}
\VerseLine{1}{Vermählungsschauder, wie er nie genossen!}
\VerseLine{1}{In Schlummer, ohn’ Berührung, hingegossen}
\VerseLine{0}{Erspüren wir der Herzen Zug und Lenken.}
\Stanza
\VerseLine{0}{Kein Tanz der Bilder vor erlosch’nem Denken.}
\VerseLine{1}{Beziehung strömt. Die Kette ist geschlossen.}
\VerseLine{1}{Durch Geistesmittel, das uns weich umflossen,}
\VerseLine{0}{Hebt Tausch nun an von Minne-Brunstgeschenken.}
\Stanza
\VerseLine{0}{Der Bann wird enger, lähmender die Schwüle —}
\VerseLine{0}{Es ist genug! die keusche Wonne tödtet!}
\VerseLine{1}{Da klirrt ein Ton! der Tag glüht ins Gemach.}
\Stanza
\VerseLine{0}{Frisch angeweht erlischt in Morgenkühle}
\VerseLine{1}{Die Geistersphäre. Bittend, völlig wach,}
\VerseLine{0}{Tret’ ich zu dir, die nun erst süss erröthet \Dots{3}}
\end{BookVerse}
\PoemEnd
