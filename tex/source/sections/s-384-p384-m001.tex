\PoemStart{s-384-p384-m001}{Der Dichter.}
\begin{BookVerse}
\VerseLine{0}{Ist dies mein Loos, so will ich mich gewöhnen}
\VerseLine{1}{Ein Nerv zu sein, der alle Schmerzen spürt,}
\VerseLine{1}{Ein Saitenspiel, so leis’ und leicht gerührt,}
\VerseLine{0}{Jedweden Lüftchens Laune auszutönen.}
\Stanza
\VerseLine{0}{Doch hört ihr’s nun in bangen Nächten stöhnen:}
\VerseLine{1}{Nicht ich —! mir hat’s der Wind dahergeführt!}
\VerseLine{0}{Ich harfe, was er singt; nicht mir gebührt}
\VerseLine{0}{Sein Lied der Trübsal schmeichelnd zu verschönen.}
\Stanza
\VerseLine{0}{Und hört ihr, lüpft der Morgen seine Schwingen,}
\VerseLine{0}{Im Silberton die starre Säule klingen,}
\VerseLine{1}{Birst sie von Glück: ihr eig’nes ist es nicht!}
\Stanza
\VerseLine{1}{Von Liebe krank, von Leben, Luft und Licht,}
\VerseLine{0}{Ein Dichter bin ich, leide mit den Dingen:}
\VerseLine{1}{Hört, was aus mir das Herz der Dinge spricht.}
\end{BookVerse}
\PoemEnd
