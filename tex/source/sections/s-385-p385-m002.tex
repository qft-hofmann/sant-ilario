\PoemStart{s-385-p385-m002}{Zwecklos.}
\begin{BookVerse}
\VerseLine{1}{Nicht nur hinauf}
\VerseLine{0}{Will Lebens Gluth und Gier.}
\VerseLine{0}{Nimmt es manch Fabelthier}
\VerseLine{1}{Doch in den Kauf!}
\Stanza
\VerseLine{0}{Kirche hat ausserm Knauf}
\VerseLine{1}{Schnörkel und Zier.}
\VerseLine{1}{Nicht nur hinauf}
\VerseLine{0}{Will Lebens Gluth und Gier.}
\Stanza
\VerseLine{0}{Menschlein und Vieh zuhauf’,}
\VerseLine{1}{Freut euch mit mir!}
\VerseLine{1}{Zwecklos wie ihr}
\VerseLine{0}{Wandl’ ich den Lebenslauf —:}
\VerseLine{0}{Nicht nur hinauf!}
\end{BookVerse}
\PoemEnd
