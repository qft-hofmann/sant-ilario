\PoemStart{s-383-p383-m002}{Unendliche Melodie.}
\begin{BookVerse}
\VerseLine{0}{Auf zitternden Flächen schreiten,}
\VerseLine{1}{Die ehernen Urtons schwingen,}
\VerseLine{1}{Zu tanzenden Weltrauch-Ringen}
\VerseLine{0}{Die Seele auswölben und weiten,}
\Stanza
\VerseLine{0}{An Kanten und Ecken und Seiten}
\VerseLine{1}{Den Blick nicht stossen und zwingen,}
\VerseLine{0}{Auf zitternden Flächen schreiten,}
\VerseLine{1}{Die ehernen Urtons schwingen,}
\Stanza
\VerseLine{0}{Kein Klammern an Einzelheiten,}
\VerseLine{1}{Ein menschentbundenes Singen,}
\VerseLine{1}{Ohn’ Ursprung quellendes Klingen,}
\VerseLine{0}{Gestaltenlos Schwimmen und Gleiten,}
\VerseLine{0}{Auf zitternden Flächen Schreiten \Dots{3}}
\end{BookVerse}
\PoemEnd
