\PoemStart{s-385-p385-m001}{An die Zeit.}
\begin{BookVerse}
\VerseLine{0}{Zeit, glänzend Meer, in deinen Gründen mein,}
\VerseLine{1}{Wo And’re nur dein Wellenspiel bestreichen!}
\VerseLine{1}{Zeit! nicht mehr Schranke den getrennten Reichen:}
\VerseLine{0}{Allgegenwart schliesst Früh’res, Künft’ges ein.}
\Stanza
\VerseLine{1}{Zukunft-Erglüh’n, Vergangenheit-Erbleichen:}
\VerseLine{0}{Ich seh’ nur grellen Mittag-Sonnenschein!}
\VerseLine{0}{Denn mir ertagte das Gewesensein}
\VerseLine{1}{Als Immerwiederkehrens Runenzeichen.}
\Stanza
\VerseLine{0}{Mag Weltgeschehen nun zum Ring sich schliessen,}
\VerseLine{0}{Mag es als Strom ins Unbegrenzte fliessen,}
\VerseLine{0}{Begrenzt vom Quell zur Mündung sich ergiessen:}
\Stanza
\VerseLine{0}{An jedem Tropfen in der Fluthen Schwalle}
\VerseLine{0}{Wirkt Wiederkunft, dass er ihr nicht entfalle,}
\VerseLine{0}{Sie sondert ihn, erstarrt ihn zum Krystalle.}
\end{BookVerse}
\PoemEnd
