\PoemStart{s-378-p378-m002}{Simplicissimus.}
\begin{BookVerse}
\VerseLine{2}{Ich liebe dich!}
\VerseLine{0}{Mag dir’s auch so zu Herzen dringen.}
\VerseLine{0}{Sonst hatt’ ich allerlei zu bringen,}
\VerseLine{2}{Heut’ bring’ ich mich.}
\Stanza
\VerseLine{0}{Sonst konnt’ ich harfen, geigen, singen,}
\VerseLine{2}{Nun heisst es: sprich!}
\VerseLine{2}{Ich liebe dich!}
\VerseLine{0}{Mag dir’s auch so zu Herzen dringen.}
\Stanza
\VerseLine{0}{Mein Töneduft, mein Farbenklingen}
\VerseLine{2}{Verscholl, erblich.}
\VerseLine{2}{Ein Dichter ich?}
\VerseLine{0}{Dem nur drei Worte noch gelingen:}
\VerseLine{2}{Ich liebe dich!}
\end{BookVerse}
\PoemEnd
