\PoemStart{s-382-p382-m002}{Apogyn.}
\begin{BookVerse}
\VerseLine{0}{Einstmals liess ich keinen Kniff}
\VerseLine{1}{Um ein Weiblein mich verdriessen.}
\VerseLine{1}{Um ein Weibesherz zu spiessen,}
\VerseLine{0}{Wie ich fein die Nadel schliff!}
\Stanza
\VerseLine{0}{Jetzo geb’ ich keinen Pfiff}
\VerseLine{1}{Auf Entbehren, auf Geniessen,}
\VerseLine{0}{Liess’ ich einstmals keinen Kniff}
\VerseLine{1}{Um ein Weiblein mich verdriessen.}
\Stanza
\VerseLine{0}{Ach, wie fern, gesehn vom Schiff,}
\VerseLine{1}{Will die Küste uns zerfliessen!}
\VerseLine{1}{Und das Weib, das wir verliessen,}
\VerseLine{0}{Dämmert bläulich zum Begriff,}
\VerseLine{0}{Einfach, ohne Pfiff und Kniff.}
\end{BookVerse}
\PoemEnd
