\PoemStart{s-375-p375-m002}{Morgenstimmung.}
\PoemSubtitle{(Nach A. Giraud.)}
\begin{BookVerse}
\VerseLine{0}{Ros’ge Stäubchen tanzen leise}
\VerseLine{1}{In des Morgens blassem Schimmer.}
\VerseLine{1}{Fernher summt es wie von schlimmer}
\VerseLine{0}{Schelmenhaft verliebter Weise.}
\Stanza
\VerseLine{0}{Schwindend nach vollbrachter Reise}
\VerseLine{1}{Sinkt der Mond, ein müder Schwimmer.}
\VerseLine{0}{Ros’ge Stäubchen tanzen leise}
\VerseLine{1}{In des Morgens blassem Schimmer.}
\Stanza
\VerseLine{0}{Husch! vor lüstern flinkem Greise}
\VerseLine{1}{Flieht ein kleines Frauenzimmer,}
\VerseLine{1}{Streift der Büsche Thaugeflimmer, —}
\VerseLine{0}{Aufgesprüht zum Wirbelkreise}
\VerseLine{0}{Ros’ge Stäubchen tanzen leise.}
\end{BookVerse}
\PoemEnd
