\PoemStart{s-381-p381-m002}{Pierrot résignant.}
\begin{BookVerse}
\VerseLine{0}{Zu welchem heil’gen Zweck — wer weiss? —}
\VerseLine{1}{Wirbt Pierrot dort um Colombine?}
\VerseLine{1}{Dass er des Weibchens Gunst verdiene,}
\VerseLine{0}{Minnt er sich müde, härmt sich heiss.}
\Stanza
\VerseLine{0}{Der Esel tanzt auf glattem Eis,}
\VerseLine{1}{Der Löwe rast vom Stich der Biene.}
\VerseLine{0}{Zu welchem heil’gen Zweck — wer weiss? —}
\VerseLine{1}{Wirbt Pierrot dort um Colombine?}
\Stanza
\VerseLine{0}{Welch Rad blieb’ ewig im Geleis,}
\VerseLine{1}{Führt ewig g’radeaus die Schiene?}
\VerseLine{1}{Da plötzlich — hellt sich seine Miene:}
\VerseLine{0}{Er geht! der Bock entsagt der Geis!}
\VerseLine{0}{Zu welchem heil’gen Zweck — wer weiss?}
\end{BookVerse}
\PoemEnd
