\PoemStart{s-378-p378-m001}{Zwei Schwäne.}
\begin{BookVerse}
\VerseLine{0}{Zwielicht des Nordens: Schimmern fahler Kerzen,}
\VerseLine{1}{Halb Licht, halb Dunkel, keines tief und ganz!}
\VerseLine{1}{Der Süden kennt des Tages vollen Glanz,}
\VerseLine{0}{Er wagt zu vollem Schwarz die Nacht zu schwärzen.}
\Stanza
\VerseLine{0}{O Südens Nacht! Beichtpriesterin der Herzen,}
\VerseLine{1}{Verschwiegenheit im heil’gen Sternenkranz!}
\VerseLine{1}{Erloschen ist der grellen Lichter Tanz,}
\VerseLine{0}{Die Stunde kam: nun taucht empor, ihr Schmerzen!}
\Stanza
\VerseLine{0}{Da sind sie, Lieb’ und Tod! Dem See entstiegen}
\VerseLine{0}{Seh’ ich zwei Schwäne schlank die Hälse biegen,}
\VerseLine{1}{Ein weisses Leuchten gleitet durch die Schatten.}
\Stanza
\VerseLine{0}{Dass wüthend sie vom See ans Ufer sprängen,}
\VerseLine{1}{Der schweren Flügel Wucht ums Haupt mir schwängen —}
\VerseLine{1}{Du Nacht des Südens wirst es nicht gestatten!}
\end{BookVerse}
\PoemEnd
